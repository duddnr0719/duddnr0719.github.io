1. 처리시간 계산

  1. 데터는 랜덤 함수를 이용하여 생성
  2. 데이터의 범위는 [1, 10,000,000]
  3. 데이터의 개수는 60,000개
  • 구현 후 각 정렬 알고리즘의 수행시간 비교(합병 정렬, 퀵 정렬, 기수 정렬 알고리즘 함께 비교)
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <string.h> 

#define N 60000
#define MIN_VAL 1
#define MAX_VAL 10000000

#define SWAP(x, y, t) ( (t)=(x), (x)=(y), (y)=(t) )

static int original_data[N];
static int list[N];
static int sorted[N];

void merge(int list[], int left, int mid, int right) {
    int i, j, k, l;
    i = left; j = mid + 1; k = left;
    
    /* 분할 정렬된 list의 합병 */
    while (i <= mid && j <= right) {
        if (list[i] <= list[j]) sorted[k++] = list[i++];
        else sorted[k++] = list[j++];
    }
    if (i > mid) { /* 남아 있는 레코드의 일괄 복사 */
        for (l = j; l <= right; l++) sorted[k++] = list[l];
    }
    else {         /* 남아 있는 레코드의 일괄 복사 */
        for (l = i; l <= mid; l++) sorted[k++] = list[l];
    }
    
    // l = left; i <= right; l++  ->  l = left; l <= right; l++
    for (l = left; l <= right; l++) list[l] = sorted[l];
}

void merge_sort(int list[], int left, int right) {
    int mid;
    if (left < right) {
        mid = (left + right) / 2;            
        merge_sort(list, left, mid);         
        merge_sort(list, mid + 1, right);    
        merge(list, left, mid, right);       
    }
}

int partition(int list[], int left, int right) {
    int pivot, temp;
    int low, high;
    
    low = left;
    high = right + 1;
    pivot = list[left];
    do {
        do 
            low++;
        while (low <= right && list[low] < pivot); 
        do 
            high--;
        while (high >= left && list[high] > pivot);
        
        if (low < high) SWAP(list[low], list[high], temp);
    } while (low < high);
    
    SWAP(list[left], list[high], temp);
    return high;
}

void quick_sort(int list[], int left, int right) {
    if (left < right) {
        int q = partition(list, left, right);
        quick_sort(list, left, q - 1);
        quick_sort(list, q + 1, right);
    }
}

#define BUCKETS 10
#define DIGITS 8 
#define MAX_QUEUE_SIZE (N + 1) 

typedef struct {
    int data[MAX_QUEUE_SIZE];
    int front, rear;
} QueueType;

void init_queue(QueueType *q) { q->front = q->rear = 0; }
int is_empty(QueueType *q) { return (q->front == q->rear); }
int is_full(QueueType *q) { return ((q->rear + 1) % MAX_QUEUE_SIZE == q->front); }

void enqueue(QueueType *q, int item) {
    if (is_full(q)) {
        return; 
    }
    q->rear = (q->rear + 1) % MAX_QUEUE_SIZE;
    q->data[q->rear] = item;
}

int dequeue(QueueType *q) {
    if (is_empty(q)) {
        return -1; // 오류
    }
    q->front = (q->front + 1) % MAX_QUEUE_SIZE;
    return q->data[q->front];
}

void radix_sort(int list[], int n) {
    int i, b, d, factor = 1;
    static QueueType queues[BUCKETS]; 
    
    for (b = 0; b < BUCKETS; b++) init_queue(&queues[b]);
    
    for (d = 0; d < DIGITS; d++) {  
        for (i = 0; i < n; i++) {
            enqueue(&queues[(list[i] / factor) % 10], list[i]);
        }
        
        for (b = i = 0; b < BUCKETS; b++) {
            while (!is_empty(&queues[b]))
                list[i++] = dequeue(&queues[b]); 
        }
        factor *= 10; 
    }
}

int main() {
    srand((int)time(NULL));
    for (int i = 0; i < N; i++) {
        original_data[i] = (rand() % MAX_VAL) + MIN_VAL;
    }

    clock_t start, end;
    double cpu_time_used;

    // --- 1. 합병 정렬 ---
    memcpy(list, original_data, N * sizeof(int));
    printf("합병 정렬...\n");
    start = clock();
    merge_sort(list, 0, N - 1);
    end = clock();
    cpu_time_used = ((double) (end - start)) / CLOCKS_PER_SEC;
    printf("  [완료] 합병 정렬 시간: %f 초\n", cpu_time_used);

    memcpy(list, original_data, N * sizeof(int));
    printf("퀵 정렬...\n");
    start = clock();
    quick_sort(list, 0, N - 1);
    end = clock();
    cpu_time_used = ((double) (end - start)) / CLOCKS_PER_SEC;
    printf("  [완료] 퀵 정렬 시간: %f 초\n", cpu_time_used);

    memcpy(list, original_data, N * sizeof(int));
    printf("기수 정렬...\n");
    start = clock();
    radix_sort(list, N);
    end = clock();
    cpu_time_used = ((double) (end - start)) / CLOCKS_PER_SEC;
    printf("  [완료] 기수 정렬 시간: %f 초\n", cpu_time_used);

    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

#define N 10
#define MIN_VAL 1
#define MAX_VAL 100
#define MAX_SIZE 100

#define SWAP(x, y, t) ( (t)=(x), (x)=(y), (y)=(t) )

int list[MAX_SIZE]; 
int n = N;

void print_array_simple(int arr[], int n, const char* title) {
    printf("%s: ", title);
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

void Print_array(int arr[], int n_total, int low_idx, int high_idx, const char* message) {
    // 배열 숫자 출력 
    for (int i = 0; i < n_total; i++) {
        printf("%4d ", arr[i]);
    }
    printf(" <- %s\n", message);
    
    // 포인터 출력
    for (int i = 0; i < n_total; i++) {
        char pointer[5] = "    "; 
        if (i == low_idx && i == high_idx) {
            strcpy(pointer, "l/h");
        } else if (i == low_idx) {
            strcpy(pointer, "low");
        } else if (i == high_idx) {
            strcpy(pointer, "high");
        }
        printf("%4s ", pointer);
    }
    printf("\n--------------------------------------------------------------\n");
}

int partition_visual(int list[], int left, int right) {
    int pivot, temp;
    int low, high;
    char msg[100]; // 메시지 버퍼
    
    low = left;
    high = right + 1;
    pivot = list[left];
    
    printf("\n[Pivot: %d (at index %d) | 범위: %d ~ %d]\n", pivot, left, left, right);
    // Hoare 파티션은 low=left, high=right+1에서 시작
    // low는 ++ 후, high는 -- 후 검사하므로 시각화는 low, high-1로 시작
    Print_array(list, n, low, high - 1, "파티션 시작");

    do {
        // low: 피벗보다 큰 값을 찾을 때까지
        do {
            low++;
            // 경계 검사 (low > right는 교차한 것이므로 제외)
            if (low <= right) 
                Print_array(list, n, low, high, "low 이동");
        } while (low <= right && list[low] < pivot);

        // high: 피벗보다 작은 값을 찾을 때까지
        do {
            high--;
             // 경계 검사 (high < left는 교차한 것이므로 제외)
            if (high >= left)
                Print_array(list, n, low, high, "high 이동");
        } while (high >= left && list[high] > pivot);
        
        if (low < high) {
            SWAP(list[low], list[high], temp);
            sprintf(msg, "swap(list[%d], list[%d])", low, high);
            Print_array(list, n, low, high, msg);
        }
    } while (low < high);
    
    Print_array(list, n, low, high, "low/high 교차");

    SWAP(list[left], list[high], temp);
    sprintf(msg, "피벗 swap(list[%d], list[%d])", left, high);
    Print_array(list, n, -1, high, msg); // high 위치에 피벗 고정

    return high;
}

void quick_sort(int list[], int left, int right) {
    if (left < right) {
        int q = partition_visual(list, left, right);
        quick_sort(list, left, q - 1);
        quick_sort(list, q + 1, right);
    }
}

int main() {
    srand((int)time(NULL));
    for (int i = 0; i < n; i++) {
        list[i] = (rand() % MAX_VAL) + MIN_VAL;
    }

    print_array_simple(list, n, "원본 데이터");
    
    quick_sort(list, 0, n - 1);
    
    printf("\n");
    print_array_simple(list, n, "최종 정렬된 데이터");

    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

#define N 10
#define MIN_VAL 0
#define MAX_VAL 100

#define BUCKETS 10
#define DIGITS 2 
#define MAX_QUEUE_SIZE (N + 1) 

void print_array(int arr[], int n, const char* title) {
    printf("%s: ", title);
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

typedef struct {
    int data[MAX_QUEUE_SIZE];
    int front, rear;
} QueueType;

void init_queue(QueueType *q) { q->front = q->rear = 0; }
int is_empty(QueueType *q) { return (q->front == q->rear); }
int is_full(QueueType *q) { return ((q->rear + 1) % MAX_QUEUE_SIZE == q->front); }

void enqueue(QueueType *q, int item) {
    if (is_full(q)) return; 
    q->rear = (q->rear + 1) % MAX_QUEUE_SIZE;
    q->data[q->rear] = item;
}

int dequeue(QueueType *q) {
    if (is_empty(q)) return -1;
    q->front = (q->front + 1) % MAX_QUEUE_SIZE;
    return q->data[q->front];
}

void print_buckets(QueueType queues[BUCKETS]) {
    printf("=======================\n");
    for (int i = 0; i < BUCKETS; i++) {
        printf("[%d]->     ", i);
        
        if (!is_empty(&queues[i])) {
            // 원형 큐의 front 다음부터 rear까지 순회
            int k = (queues[i].front + 1) % MAX_QUEUE_SIZE;
            while (1) {
                printf("%3d ", queues[i].data[k]); 
                if (k == queues[i].rear) break; // rear에 도달하면 종료
                k = (k + 1) % MAX_QUEUE_SIZE;
            }
        }
        printf("\n");
    }
    printf("=======================\n");
}

void radix_sort(int list[], int n) {
    int i, b, d, factor = 1;
    QueueType queues[BUCKETS];
    
    for (b = 0; b < BUCKETS; b++) init_queue(&queues[b]);
    
    for (d = 0; d < DIGITS; d++) {  
        // 데이터들을 자리수에 따라 큐에 삽입
        for (i = 0; i < n; i++) {
            enqueue(&queues[(list[i] / factor) % 10], list[i]);
        }
        
        printf("\n--- 자릿수 %d (Pass %d) 기준 정렬 ---\n", factor, d + 1);
        print_buckets(queues);

        // 버킷에서 꺼내어 list로 합침
        for (b = i = 0; b < BUCKETS; b++) {
            while (!is_empty(&queues[b]))
                list[i++] = dequeue(&queues[b]); 
        }
        factor *= 10; // 그 다음 자리수로 이동
    }
}

int main() {
    int list[N]; 
    
    // 0-100 범위의 랜덤 데이터 10개 생성
    srand((int)time(NULL));
    for (int i = 0; i < N; i++) {
        list[i] = rand() % (MAX_VAL + 1);
    }

    print_array(list, N, "원본 데이터");
    
    radix_sort(list, N);
    
    printf("\n");
    print_array(list, N, "최종 정렬된 데이터");

    return 0;
}
