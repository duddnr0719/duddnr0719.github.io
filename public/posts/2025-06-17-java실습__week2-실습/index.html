소수 판별 프로그램

판별을 원하는 수를 num 변수에 저장하고 프로그램을 실행하여 소수인지 아닌지 판별.

public class Prime {
public static void main(String args[]) {
int num = 30; // 소수인지 아닌지 판별할 숫자
boolean flag = false;
for (int i = 2; i < =num;) {
if (num%1 == 0) flag = true; // 소수가 아닌 경우 판별
break;
}
if (flag == true) System.out.println(num + "은 소수가 아닙니다.");
else System.out.println(num + "은 소수입니다.");
}
}

import java.util.*;

할 일 관리 프로그램

사용자가 할 일 목록을 입력하고, 완료한 항목을 체크하는 Todo 리스트 프로그램

import java.util.*;

public class TodoList {
public static ArrayList<String> todolist = new ArrayList<>();

private static void add(Scnaner scanner) {
System.out.print("추가할 일: ");
String item = scanner.nextLine();
todolist.add(item);
System.out.println("추가 되었습니다.");
}

private static void show() {
if (todolist.isEmpty() {
System.out.println("할 일 목록이 비었습니다.");
return;
}
else {
System.out.println("할 일 목록: ");
for (int i = 0; i < todolist.size(); i++) System.out.println((i+1) + ". " + todolist.get(i));
}
}

private static void remove(Scanner scanner) {
show();
System.out.print("완료된 항목의 번호: ");
int choice = scanner.nextInt();
scanner.nextLine();

if (choice > 0 && choice <= todolist.size()) {
todolist.remove(choice-1);
System.out.println("항목이 삭제되었습니다.);
return;
} else System.out.println("잘못된 번호입니다.");
}

public static void main(String args[]) {
Scanner scanner = new Scanner(System.in);

while(flag) {
System.out.println("ToDo 리스트");

int choice = scanner.nextInt();
scanner.nextLine();

switch(choice) {

add(scanner);

show();
break;

tmp = false;

숫자 맞추기 프로그램

랜덤 정수를 뽑아 사용자가 그 정수를 맞출 때까지 실행되는 프로그램

import java.util.*;

public class GuessNumber {
public static void main(String args[]) {
Scanner sc = new Scanner(System.in);
Random rd = new Random();

int maxnum = 100;
int randomnum = rd.nextInt(maxnum) + 1;
int choice = 0;

System.out.println("1부터" + maxnum + "까지 숫자 중 한 개를 선택하세요. ");

while (choice != randomnum) {
System.out.print("숫자를 입력하세요: ");
choice = sc.nextInt();
sc.nextLine();
if (choice < randomnum) System.out.println("더 큰 수입니다.");
else if (choice > randomnum) System.out.println("더 작은 수입니다.");
else System.out.println("정답입니다.");
}
sc.close();
}
}

계산기 프로그램

사용자로부터 피연잔자와 연산자를 입력받아 연산자를 기준으로 조건문을 실행하여 피연산자를 계산하여 출력.

import java.util.*;

public class Calculator {
public static void main(String args[]) {
Scanner sc = new Scanner(System.in);

System.out.print("연산자를 입력하세요.(+, -, /, *): ");
String op = sc.next();

System.out.print("피연산자 두 개를 입력하세요: ");
int a = sc.nextInt();
int b = sc.nextInt();
int c;
if (op.equals("+")) {
c = a + b;
System.out.println(c);
}
else if (op.equals("-")) {
c = a - b;
System.out.println(c)
}
else if (op.equals("*")) {
c = a * b;
System.out.println(c);
}
else if (op.equals("/")) {
c = a / b;
System.out.println(c);
}
sc.close();
}
}
