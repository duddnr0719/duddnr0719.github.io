9. 그래프 I

2025년 10월 24일 데이터구조II

그래프

  • 객체 사이의 연결 관계를 표현할 수 있는 자료 구조. → ex) 지도
  • 그래프 G는 (V, E)로 표시
  • 정점(Vertices)
  • 여러가지 특성을 가질 수 있는 객체를 의미한다.
  • V(G) : 그래프 G의 정점들의 집합
  • 노드(node)라고도 불림.
  • 간선(edge)
  • 정점들 간의 관계를 의미한다.
  • E(G) : 그래프 G의 간선들의 집합
  • 링크(link)라고도 불림.
## 무방향 그래프 - 간선을 통해서 양방향으로 갈 수 있음을 나타낸다. - 정점 A와 정점 B를 연결하는 간선은 (A, B)와 같이 소괄호를 이용하여 표현한다. → (A, B) / (B, A)는 동일한 간선이다. ![](/images/b99fb78ad3d22d97.png) ## 방향 그래프 - 간선에 방향성이 존재하는 그래프 → 도로의 일방통행길처럼 간선을 통하여 한쪽 방향으로만 갈 수 있음을 나타낸다. - 정점 A와 정점 B로만 갈 수 있는 간선은 \로 표시한다. → \ / \는 서로 다른 간선이다. ![](/images/79fdf340a707a366.png) ## 네트워크(가중치 그래프) - 간선에 비용이나 가중치가 할당된 그래프 ![](/images/6c8f4bcc348df8dc.png) ## 부분 그래프 - 어떤 그래프의 정점의 일부와 간선의 일부로 이루어진 그래프 ![](/images/e5b11beabdbcd139.png) ## 인접 정점 - 하의 정점에서 간선에 의해 직접 연결된 정점 ## 무방향 그래프의 차수 - 하의 정점에 연결된 다른 정점의 수 ## 방향 그래프의 차수 - 진입 차수 : 외부에서 오는 간선의 수 - 진출 차수 : 외부로 향하는 간선의 수 - 방향 그래프의 모든 집입/진출 차수의 합은 간선의 수. ## 단순 경로 - 경로 중에서 반복되는 간선이 없는 경로 ## 사이클 - 단순 경로의 시작 정점과 종료 정점이 동일한 경로

그래프 ADT

- 객체 : 정점의 집합과 간선의 집합
- 연산 : 

- create_graph() ::= 그래프를 생성한다.
- init(g) ::= 그래프 g를 초기화한다.
- insert_vertext(g, v) ::= 그래프 g에 정점 v를 삽입한다.
- insert_edge(g, u, v) ::= 그래프 g에 간선 (u, v)를 삽입한다.
- delete_vertext(g, v) ::= 그래프 g의 정점 v를 삭제한다.
- delete_edge(g, u, v) ::= 그래프 g의 간선 (u, v)를 삭제한다.
- is_empty(g) ::= 그래프 g가 공백 상태인지 확인한다.
- adjacent(v) ::= 정점 v에 인접한 정점들의 리스트를 반환한다.
- destroy_graph(g) ::= 그래프 g를 제거한다.

그래프 표현 방법

  • 인접 행렬(adjacent matrix) 방법 → 메모리 낭비가 심해 잘 사용하지 않음.
  • 인접 리스트(adjacent list) 방법

인접 행렬 방법

인접 행렬 구현

#include <stdio.h>
#include <stdlib.h>

#define MAX_VERTICES 50

typedef struct GraphType {
    int n;
    int adj_mat[MAX_VERTICES][MAX_VERTICES];
} GraphType;

// 그래프 초기화
void init(GraphType* g) {
    int r, c;
    g->n = 0;
    for (r = 0; r < MAX_VERTICES; r++) {
        for (c = 0; c < MAX_VERTICES; c++) {
            g->adj_mat[r][c] = 0;
        }
    }
}

void insert_vertex(GraphType* g, int v) {
    if (((g->n) + 1) > MAX_VERTICES) {
        fprintf(stderr, "그래프 : 정점의 개수 초과");
        return;
    }
    g->n++;
}

void insert_edge(GraphType* g, int start, int end) {
    if (start >= g->n || end >= g->n) {
        fprintf(stderr, "그래프 : 정점 번호 오류");
        return;
    }
    g->adj_mat[start][end] = 1;
    g->adj_mat[end][start] = 1;
}

int main(void) {
    GraphType *g;
    g = (GraphType*)malloc(sizeof(GraphType));
    init(g);
    for (int i = 0; i < 4; i++) {
        insert_vertex(g, i);
    }

    insert_edge(g, 0, 1);
    insert_edge(g, 0, 2);
    insert_edge(g, 0, 3);
    insert_edge(g, 1, 2);
    insert_edge(g, 2 ,3);

    free(g);
}

인접 리스트 방법

  • 각 정점에 인접한 정점들을 연결 리스트로 표현 ( 밑 예시는 무방향 그래프 )

인접 리스트 구현

#include <stdio.h>
#include <stdlib.h>

#define MAX_VERTICES 50


typedef struct GraphNode {
    int vertex;                  
    struct GraphNode* link;      
} GraphNode;


typedef struct GraphType {
    int n;                       
    GraphNode* adj_list[MAX_VERTICES];
} GraphType;

// 그래프 초기화
void init(GraphType* g) {
    g->n = 0;
    for (int i = 0; i < MAX_VERTICES; i++) g->adj_list[i] = NULL;
}

// 정점 추가
void insert_vertex(GraphType* g, int v) {
    if (((g->n) + 1) > MAX_VERTICES) {
        fprintf(stderr, "그래프 : 정점의 개수 초과");
        return;
    }
    g->n++;
}

// 간선 추가 (무방향)
void insert_edge(GraphType* g, int start, int end) {
    if (start >= g->n || end >= g->n) {
        fprintf(stderr, "그래프 : 정점 번호 오류");
        return;
    }

    GraphNode* node = (GraphNode*)malloc(sizeof(GraphNode));
    node->vertex = end;
    node->link = g->adj_list[start];
    g->adj_list[start] = node;

// 만약 단방향 그래프라면 이 부분을 제거한다.
    node = (GraphNode*)malloc(sizeof(GraphNode));
    node->vertex = start;
    node->link = g->adj_list[end];
    g->adj_list[end] = node;
}

int main(void) {
    GraphType *g;
    g = (GraphType*)malloc(sizeof(GraphType));
    init(g);
    for (int i = 0; i < 4; i++) {
        insert_vertex(g, i);
    }

    insert_edge(g, 0, 1);
    insert_edge(g, 0, 2);
    insert_edge(g, 0, 3);
    insert_edge(g, 1, 2);
    insert_edge(g, 2, 3);

    free(g);
}

그래프 탐색

- 깊이 우선 탐색(DFS : Depth First Search) → 한 방향으로 갈 수 있을 때까지 가다가 더 이상 갈 수 없게 되면 가장 가까운 갈림길로 돌아와서 이 곳으로부터 다른 방향으로 다시 탐색 진행 → 되돌아가기 위해서 스택 필요 (⇒ 순환호출을 사용하여 묵시적 스택 이용) ![](/images/9244547a57e29e7b.png)

코드

// 인접 행렬로 표현된 그래프에 대한 깊이 우선 탐색
void dfs_mat(GraphType* g, int v) {
    int w;
    visited[v] = TRUE;
    printf("정점 %d -> ", v);
    for (w = 0; w < g->n; w++) {
        if (g->adj_mat[v][w] && !visited[w]) {
            dfs_mat(g, w);
        }
    }
}

// 인접 리스트로 표현된 그래프에 대한 깊이 우선 탐색
void dfs_list(GraphType* g, int v) {
    GraphNode* w;
    visited[v] = TRUE;
    printf("정점 %d -> ", v);
    for (w = g->adj_list[v]; w; w = w->link) {
        if (!visited[w->vertex]) {
            dfs_list(g, w->vertex);
        }
    }
}

## 시간 복잡도 - 인접 행렬 : O(n\^2) - 인접 리스트 : O(n+e) → n : 정점의 개수 , e : 간선의 개수
