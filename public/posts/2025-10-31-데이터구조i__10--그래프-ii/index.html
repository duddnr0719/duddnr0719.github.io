신장 트리

  • 그래프 내의 모든 정점 포함하는 트리 (→ cycle이 생기면 안됨)
  • n개의 정점을 가지는 그래프의 신장 트리는 n-1개의 간선을 가짐

최소 비용 신장 트리(minimum spanning tree)

- 네트워크에 있는 모든 정점들을 가장 적은 수의 간선과 비용으로 연결 ![](/images/1589d6954d5cd95d.png) ## Kruskal의 MST 알고리즘 - 탐욕적인 방법(greedy) - 주요 알고리즘 설계 기법 - 각 단계에서 최선의 답을 선택하는 과정을 반복함으로써 최종적인 해답에 도달 - 탐욕적인 방법은 항상 최적의 해답을 주는지 검증 필요 - Kruskal MST 알고리즘은 최적의 해답임이 증명됨 - union-find 알고리즘 → 원소가 어떤 집합에 속하는지 알아냄 → Kruskal MST 알고리즘에서 사이클 검사에 사용 (⇒ 신장 트리는 사이클이 생성되면 안되기 때문) ![](/images/8a970b7d135535f9.png) ![](/images/ec7acc3d013aedd4.png) ## Kruskal MST 코드 구현 ```c #include #include

    e = g->edges[i];

    vset = set_find(e.end);                 // 정점 v의 집합 번호
    if (uset != vset) {                     // 서로 속한 집합이 다른 경우
        printf("간선 (%d, %d) %d 선택\n", e.start, e.end, e.weight);
        edge_accepted++;
        set_union(uset, vset);              // 두 개의 집합을 합침
    }
    i++;
}

## 시간 복잡도
- Kruskal 알고리즘은 대부분 간선들을 정렬하는 시간에 좌우됨
  - 사이클 테스트 등의 작업은 정렬에 비해 매우 신속하게 수행됨
- 네트워크의 간선 e개를 퀵정렬과 같은 효율적인 알고리즘으로 정렬한다면 Kruskal 알고리즘의 시간 복잡도는 O(e*log(e))가 된다.

## Prim MST 알고리즘 코드 구현
```c
#include <stdio.h>
#include <stdlib.h>

#define TRUE 1
#define FALSE 0
#define MAX_VERTICES 100
#define INF 1000L

typedef struct GraphType {
	int n;	// 정점의 개수
	int weight[MAX_VERTICES][MAX_VERTICES];
} GraphType;

int selected[MAX_VERTICES];
int distance[MAX_VERTICES];

// 최소 dist[v] 값을 갖는 정점을 반환
int get_min_vertex(int n)
{
	int v, i;
	for (i = 0; i <n; i++)
		if (!selected[i]) { // 정점을 방문하지 않은 경우
			v = i;
			break;
		}
	for (i = 0; i < n; i++)
		if (!selected[i] && (distance[i] < distance[v])) v = i; // 정점을 방문하지 않고 간선의 거리가 더 짧은 경우
	return v;
}
// Prim의 최소 비용 신장 트리 프로그램
void prim(GraphType* g, int s)
{
	int i, u, v;

	for (u = 0; u<g->n; u++)	distance[u] = INF; // distance 배열 초기화
	distance[s] = 0; // 시작 인덱스의 거리 0으로 설정 -> 시작점부터 시작점까지의 거리는 0.
	for (i = 0; i<g->n; i++) {
		u = get_min_vertex(g->n); // 방문하지 않은 정점 중 거리가 가장 짧은 정점의 인덱스 값 반환 받음.
		selected[u] = TRUE; // 해당 정점 방문 표시
		if (distance[u] == INF) return;
		printf("정점 %d 추가\n", u);
		for (v = 0; v<g->n; v++)
			if (g->weight[u][v] != INF)
				if (!selected[v] && g->weight[u][v]< distance[v]) // 방문이 되지 않은 정점 중 거리가 짧은 간선이 존재하는 경우 실행
					distance[v] = g->weight[u][v]; // 거리가 짧은 간선의 값으로 distance 배열의 값 변경
	}
}

### Floyd 알고리즘
<columns>
<column ratio="100">- 모든 정점에서 다른 모든 정점까지의 최단 경로를 계산
- 2차원 배열 A를 이용하여 3중 반복을 하는 루프로 구성됨
- 0부터 k까지의 정점만을 이용하여 정점 i부터 정점 j까지의 최단 경로 길이
  - 이에 해당하는 2가지 경우
    1. 정점 k를 거치지 않는 경우
    → 배열 A는 k보다 큰 정점은 통과하지 않으므로 최단 거리는 동일
    1. 정점 k를 거치는 경우
    → i에서 k까지의 최단거리에 k에서 j까지의 최단 거리를 더한 값</column>

## Floyd 최단경로 알고리즘 프로그램
```c
#include <stdio.h>
#include <stdlib.h>

#define TRUE 1
#define FALSE 0
#define MAX_VERTICES	100	
#define INF	1000000	/* 무한대 (연결이 없는 경우) */

typedef struct GraphType {
	int n;	// 정점의 개수
	int weight[MAX_VERTICES][MAX_VERTICES];
} GraphType;

int A[MAX_VERTICES][MAX_VERTICES];

void printA(GraphType *g)
{
	int i, j;
	printf("===============================\n");
	for (i = 0; i<g->n; i++) {
		for (j = 0; j<g->n; j++) {
			if (A[i][j] == INF)
				printf("  * ");
			else printf("%3d ", A[i][j]);
		}
		printf("\n");
	}
	printf("===============================\n");
}

void floyd(GraphType* g)
{

	int i, j, k;
	for (i = 0; i<g->n; i++)
		for (j = 0; j<g->n; j++)
			A[i][j] = g->weight[i][j];
	printA(g);

	for (k = 0; k<g->n; k++) {
		for (i = 0; i<g->n; i++)
			for (j = 0; j<g->n; j++)
				if (A[i][k] + A[k][j] < A[i][j])
					A[i][j] = A[i][k] + A[k][j];
		printA(g);
	}
}

int main(void)
{
	GraphType g = { 7,
	{{ 0,  7,  INF, INF,   3,  10, INF },
	{ 7,  0,    4,  10,   2,   6, INF },
	{ INF,  4,    0,   2, INF, INF, INF },
	{ INF, 10,    2,   0,  11,   9,   4 },
	{ 3,  2,  INF,  11,   0, INF,   5 },
	{ 10,  6,  INF,   9, INF,   0, INF },
	{ INF, INF, INF,   4,   5, INF,   0 } }
	};
	floyd(&g);
	return 0;
}

시간 복잡도

  • 네트워크에 n개의 정점이 있다면 Floyd의 최단경로 알고리즘은 3중 반복문을 실행하므로 O(n^3)이다.
  • 모든 정점 상의 최단경로를 구하려면 Dijkstra의 알고리즘을 n번 반복해도 되며, 이 경우 전체 복잡도는 O(n^3)이 된다.

위상 정렬(Topological Sort)

- 방향 그래프에서 간선 가 있다면 정점 u는 정점 v를 선행한다. - 방향 그래프 정점들의 선행 순서를 위배하지 않으면서 모든 정점을 나열한다. ![](/images/4e2d333dcb8d59aa.png) --- - 진입 차수의 비 내림차순 순서로 정렬을 한다. - 진입 차수가 0개 → n개 순으로 탐색하고, 정렬 과정을 거치며 진입 차수가 0인 노드들은 제거하고, 진입 차수가 0이 아닌 노드들은 점점 0으로 수정한 뒤 제거하며 정렬을 한다. - 진입 차수 : 해당 노드로 들어오는 간선의 개수 ⇒ A→B, C→B 와 같은 형태의 그래프라면 A,C의 진입 차수 : 0, B의 진입 차수 : 2 ![](/images/e84e67cd0be00a25.png) ## 위상 정렬 프로그램 ```c #include #include #define TRUE 1 #define FALSE 0 #define MAX_VERTICES 50
