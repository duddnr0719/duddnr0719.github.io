5. 최소비용 신장트리

2025년 11월 10일 데이터구조 실습

1. 개요

가. Kruskal의 MST 를 구한다. 나. Prim의 MST를 구하고 지난 시간 구현한 Kruskal의 MST과 비교한다.

  1. 각 지역간의 거리는 다음과 같다.
  2. Kruskal MST 출력
  3. 위의 그래프에서 Kruskal 알고리즘으로 최소 비용 신장 트리를 생성하는 과정을 출력하여라. 1. 각 지역명을 숫자로 변환하여 구현한다. (예:인천은 0, 서울은 1,…전주는 4, 대전은 5,… ) 1. MST 출력이 종료되면 마지막에 최종결과로 MST 간선의 가중치의 합을 출력한다.
  • 출력하는 화면은 다음과 같다. (0, 1) (4, 5) … 933
#include <stdio.h>
#include <stdlib.h>

#define TRUE 1
#define FALSE 0
#define MAX_VERTICES 100
#define INF 1000

int parent[MAX_VERTICES];

// 집합 초기화
void set_init(int n) {
    for (int i = 0; i < n; i++) {
        parent[i] = -1;
    }
}

// curr가 속하는 집합의 루트를 찾아서 반환
int set_find(int curr) {
    if (parent[curr] == -1) return curr;
    while (parent[curr] != -1) curr = parent[curr];
    return curr;
}

// 두 원소가 속한 집합을 합침
void set_union(int a, int b) {
    int root1 = set_find(a);
    int root2 = set_find(b);
    if (root1 != root2) parent[root1] = root2;
}

struct Edge {
    int start, end, weight;
};

typedef struct GraphType {
    int n_e; // 간선의 개수 (number of edges)
    int v;   // 정점의 개수 (number of vertices)
    struct Edge edges[2 * MAX_VERTICES];
} GraphType;

// 그래프 초기화
void graph_init(GraphType* g, int num_vertices) {
    g->n_e = 0;
    g->v = num_vertices;
    for (int i = 0; i < 2 * MAX_VERTICES; i++) {
        g->edges[i].start = 0;
        g->edges[i].end = 0;
        g->edges[i].weight = INF;
    }
}

// 간선 삽입 연산
void insert_edge(GraphType* g, int start, int end, int w) {
    g->edges[g->n_e].start = start;
    g->edges[g->n_e].end = end;
    g->edges[g->n_e].weight = w;
    g->n_e++; // 간선 개수 증가
}

// qsort()에 사용되는 비교 함수
int compare(const void* a, const void* b) {
    struct Edge* x = (struct Edge*)a;
    struct Edge* y = (struct Edge*)b;
    return (x->weight - y->weight);
}

// kruskal의 최소 비용 신장 트리 프로그램
void kruskal(GraphType* g) {
    int edge_accepted = 0;
    int uset, vset;
    struct Edge e;
    int total_weight = 0; // 가중치 합계를 위한 변수

    // 정점의 개수 기준으로 집합 초기화
    set_init(g->v);
    // 간선의 개수 기준으로 정렬
    qsort(g->edges, g->n_e, sizeof(struct Edge), compare);

    printf("크루스칼 최소 신장 트리 알고리즘 \n");
    int i = 0;
    // (정점의 개수 - 1)개의 간선을 선택할 때까지 반복
    while (edge_accepted < (g->v - 1)) {
        e = g->edges[i];
        uset = set_find(e.start);
        vset = set_find(e.end);
        if (uset != vset) { // 사이클을 형성하지 않으면
            printf("간선 (%d, %d) %d 선택\n", e.start, e.end, e.weight);
            edge_accepted++;
            set_union(uset, vset);
            total_weight += e.weight; // 선택된 간선의 가중치를 더함
        }
        i++;
    }
    printf("가중치 총합 : %d \n", total_weight);
}

int main(void)
{
    GraphType *g;
    g = (GraphType *)malloc(sizeof(GraphType));
    // 정점은 0~8번까지 총 9개
    graph_init(g, 9);

    // 인천 : 0, 서울 : 1, 원주 : 2, 강릉 : 3, 전주 : 4, 대전 : 5, 대구 : 6, 부산 : 7, 광주 : 8
    insert_edge(g, 0, 1, 35);
    insert_edge(g, 1, 2, 123);
    insert_edge(g, 2, 3, 117);
    insert_edge(g, 0, 4, 245);
    insert_edge(g, 4, 5, 87);
    insert_edge(g, 1, 5, 154);
    insert_edge(g, 5, 2, 165);
    insert_edge(g, 5, 6, 155);
    insert_edge(g, 6, 2, 222);
    insert_edge(g, 6, 7, 164);
    insert_edge(g, 4, 8, 98);
    insert_edge(g, 8, 5, 122);

    kruskal(g);
    
    free(g);
    return 0;
}

4. Prim MST

  1. 위의 그래프에서 Prim 알고리즘으로 최소 비용 신장 트리를 생성하는 과정을 출력하여라. (시작위치는 서울이다)
  2. 각 지역명을 숫자로 변환하여 구현한다. (예:인천은 0, 서울은 1,…전주는 4, 대전은 5,… )
  3. MST 출력이 종료되면 마지막에 최종결과로 MST 간선의 가중치의 합을 출력한다.
  4. selected[]와 distance[]값의 변화를 출력하라.
#include <stdio.h>
#include <stdlib.h>

#define TRUE 1
#define FALSE 0
#define MAX_VERTICES 100
#define INF 1000L

typedef struct GraphType {
    int n;  // 정점의 개수
    int weight[MAX_VERTICES][MAX_VERTICES];
} GraphType;

int selected[MAX_VERTICES];
int distance[MAX_VERTICES];

// 최소 dist[v] 값을 갖는 정점을 반환
int get_min_vertex(int n)
{
    int v, i;
    for (i = 0; i <n; i++)
        if (!selected[i]) {
            v = i;
            break;
        }
    for (i = 0; i < n; i++)
        if (!selected[i] && (distance[i] < distance[v])) v = i;
    return (v);
}

// 각 단계별 배열 상태를 출력하는 함수
void status(GraphType* g) {
    int step = 0;
    printf("--- [STEP %d] ---\n", step++);
    printf(" selected: ");
    for (int i = 0; i < g->n; i++) {
        printf("%3d", selected[i]);
    }
    printf("\n");
    printf(" distance: ");
    for (int i = 0; i < g->n; i++) {
        if (distance[i] == INF)
            printf(" INF");
        else
            printf("%4d", distance[i]);
    }
    printf("\n\n");
}

void prim(GraphType* g, int s) {
    int i, u, v;
    int total_distance = 0;
    int visit[MAX_VERTICES];
    int index = 0;

    for (u = 0; u<g->n; u++)
        distance[u] = INF;
    distance[s] = 0;
    for (i = 0; i<g->n; i++) {
        u = get_min_vertex(g->n);
        selected[u] = TRUE;
        if (distance[u] == INF) return;
        visit[index++] = u;
        total_distance += distance[u];
        status(g);
        for (v = 0; v<g->n; v++) {
            if (g->weight[u][v] != INF) {
                if (!selected[v] && g->weight[u][v]< distance[v]) {
                    distance[v] = g->weight[u][v];
                    
                }
            }
            
        }
    }
    for (int i = 0; i < index; i++) {
        printf("%d 추가 -> ", visit[i]);
    }
    printf("\n");
    printf("가중치 총합 : %d\n", total_distance);
    
}

int main(void)
{

    // 인천 : 0, 서울 : 1, 원주 : 2, 강릉 : 3, 전주 : 4, 대전 : 5, 대구 : 6, 부산 : 7, 광주 : 8
    GraphType g = { 9, 
    {{ 0, 35, INF, INF, 245, INF, INF, INF, INF},
    { 35, 0, 123, INF, INF, 154, INF, INF, INF},
    { INF, 123, 0, 117, INF, 165, 222, INF, INF},
    { INF, INF, 117, 0, INF, INF, INF, INF, INF},
    { 245, INF, INF, INF, 0, 87, INF, INF, 98},
    { INF, 154, 165, INF, 87, 0, 155, INF, 122},
    { INF, INF, 222, INF, INF, 155, 0, 164, INF},
    { INF, INF, INF, INF, INF, 164, 0, INF, INF},
    { INF, INF, INF, INF, 98, 122, INF, INF, 0}} 
    };

    prim(&g, 1);
    return 0;
}
