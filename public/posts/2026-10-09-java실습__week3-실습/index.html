클래스 정의와 메소드 사용 예제

Counter 클래스에 생성자, 접근자, 설정자를 만들고 up(), down(), reset() 메소드를 만든다. 두 번의 up() 메소드, 한 번의 down() 메소드를 실행한 후 getCount()로 카운터 값을 추출하여 출력하는 프로그램을 작성하라.

class CounterTest {
private int counter;

public CounterTest(int counter) {
this.counter = counter;
}

public void up() counter++;
public void down() counter--;
public void reset() counter = 0;

public int getCount() return counter;
public void setCount(int counter) {
this.counter = counter;
}
}
public class Counter {
public static void main(String args[]) {
CounterTest counter = new CounterTest(0);
counter.reset();

System.out.println("객체 생성후 Counter Value: " + counter.getCount());
counter.up();
System.out.println("up() 메소드 호출 후 Counter Value: " + counter.getCount());
counter.up();
System.out.println("up() 메소드 호출 후 Counter Value: " + counter.getCount());
counter.down();
system.out.println("down() 메소드 호출 후 Counter Value: " + counter.getCount());
}
}

ATM 시스템

은행 계좌를 관리하는 시스템. Account 클래스를 만들고 계좌 잔액, 소유자 이름 등을 저장하는 객체를 생성하여 은행 계좌 시스템을 구현하라. 입금, 출금, 잔액 조회 등의 기능을 추가한다.

public class Account {
private double balance;

public Account(double balance) {
this.balance = balance;
}

public void deposit(double amount) {
if (amount > 0) { 
balance += amount;
System.out.println(amount + "원이 입금 되었습니다.");
}
else {
System.out.println("올바른 값을 입력하세요.");
}
}
public void withdraw(double amount) {
if (amount > 0 && amount <= balance) {
balance -= amount;
System.out.println(amount + "원이 출금 되었습니다.");
}
else if (amount > balance) System.out.println("잔액이 부족합니다");
else System.out.println("올바른 값을 입력하세요.");
}
public double getBalance() return balance;
}

public class Atm {
public static void main(String args[]) {
Scanner sc = new Scanner(System.in);
Account account = new Account(0);

boolean flag = true;

while(flag) {
System.out.println("===== 은행 계좌 시스템 =====");
System.out.println("1. 입금");
System.out.println("2. 출금");
System.out.println("3. 잔액 조회");
System.out.println("4. 종료");
System.out.print("메뉴 선택: ");
int choice = sc.nextInt();
sc.nextLine();

switch(choice) {
case 1: 
System.out.print("입금할 금액: ");
double depositAmount = sc.nextDouble();
account.deposit(depositAmount);
break;
case 2:
System.out.println("출금할 금액: ");
double withdrawAmount = sc.nextDouble();
account.withdraw(withdrawAmount);
break;
case 3:
System.out.println("현재 잔액: " + account.getBalance() + "원");
break;
case 4:
scanner.close();
flag = false;
break;
default:
System.out.println("올바른 번호를 입력하세요.");
}
}
}
}

원 다루는 프로그램

Circle 클래스를 정의하고 원의 중심을 나타내는 x, y 필드를 추가한다. 원의 색상을 나타내는 color도 추가한다. move() 메소드를 추가하여 원의 중심을 랜덤하게 이동시킨다. main() 메소드에서 3개의 원을 생성하고 이들을 100번 랜덤하게 이동시키는 코드를 작성하라. 각 원들의 반지름은 서로 다르게 한다.

class Circle {
private int x, y, radius;
private String color;
private int max = 500;

public CircleTest(int x, int y, int radius, String color) {
this.x = x;
this.y = y;
this.radius = radius;
this.color = color;
}
public void move() {
Random rd = new Random();
for (int i = 0; i < 100; i++) {
x = rd.nextInt(max);
y = rd.nextInt(max);
System.out.println("이동한 x,y 좌표: " + x + ", "+ y);
}
}
public int getX() return x;
public int getY() return y;
public int getRadius() return radius;
public String getColor() return color();
}

public class CircleTest {
public static void main(String args[]) {
Circle c1 = new Circle(0, 0, 100, "red");
Circle c2 = new Circle(0, 0, 20, "blue");
Circle c3 = new Circle(0, 0, 50, "green");

c1.move();
c2.move();
c3.move();
}
}
