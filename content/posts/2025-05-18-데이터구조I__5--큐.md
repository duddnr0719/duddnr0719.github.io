# 큐 ?
> 선입선출(FIFO : First In First Out)의 형태를 가진 자료구조. 먼저 들어온 데이터가 먼저 나간다.
매표소나 계산대의 대기열과 같다.
## 큐 ADT
```c
- 객체 : 0개 이상의 요소들로 구성된 선형 리스트
- 연산 :

// 큐 생성
create(max_size) ::= 최대 크기가 max_size인 공백큐를 생성한다.
// 큐 초기화
init(q) ::= 
if(size == 0) return TRUE;
else return FALSE;
// 공백 상태 검출 함수
is_empty(q) ::=
if(size == 0) return TRUE;
else return FALSE;
// 포화 상태 검출 함수 
is_full(q) ::=
if (size == max_size) return TRUE;
else return FALSE;
// 삽입 함수
enqueue(q, e) ::=
if(is_full(q)) err;
else q의 끝에 e를 추가한다.
// 제거 함수
dequeue(q) ::=
if(is_empty(q)) err;
else q의 맨 앞에 있는 값을 제거하여 반환한다.
// 피크 함수
peek(q) ::=
if(is_empty(q)) err;
else q의 맨 앞에 있는 값을 읽어서 반환한다.
```
<callout>
큐의 연산에서 삽입과 삭제는 한 쪽에서만 이루어지는 스택과 달리 양 쪽에서 이루어질 수 있다. 따라서 변수가 두 개 사용된다.
→ 삽입에 사용되는 변수 : rear(큐의 마지막 요소) / 삭제에 사용되는 변수 : front(큐의 첫 번째 요소)
</callout>

## 선형큐(linear queue)
![](/images/9fe0312a7f4c1b69.png)
1차원 배열을 사용하여 구현한 큐. 초기값은 -1이고 데이터가 증가되면 rear를 하나 증가하고 그 위치에 데이터가 저장된다. 삭제할 때는 front를 하나 증가하고 그 위치에 있는 데이터를 삭제한다.

### 선형큐 구현
```c
#include <stdio.h>
#include <stdlib.h>
#define MAX_SIZE 5

typedef int element;
typedef struct {
int front;
int rear;
element data[MAX_SIZE];
}Queue;
// 오류 함수
void error(char *message) {
fprintf(stderr, "%s\n", message);
exit(1);
}
void init_queue(Queue *q) {
q->rear = -1;
q->front = -1;
}
void queue_print(Queue *q) {
for (int i = 0; i < MAX_SIZE; i++) {
if (i <= q->front || i > q->rear) printf("|");
else printf("%d | ", q->data[i];
}
printf("\n");
}
int is_full(Queue *q) {
if (q->rear == MAX_SIZE-1) return 1;
else return 0;
}
int is_empty(Queue *q) {
if (q->front == q->rear) return 1;
else return 0;
}
void enqueue(Queue *q, int item) {
if (is_full(q)) {
error("queue if full");
return;
}
q->data[++(q->rear)] = item;
}
int dequeue(Queue *q) {
if(is_empty(q)) {
error("queue is empty");
return -1;
}
int item = q->data[++(q->front)];
return item;
}
```

### 선형큐의 문제점
front와 rear의 값이 계속 증가만 하기 때문에 사용을 하다보면 배열의 끝에 도달하게 되고, 배열의 앞부분이 비어 있더라도 사용하지 못한다.
→ 주기적으로 모든 요소들을 왼쪽으로 이동시켜야하는 문제점이 생긴다. (이런 식으로 요소들을 이동시키면 상당한 시간과 코딩이 복잡해진다.)

## 원형큐
![](/images/48dc3c81dbbcc844.png)
→ 선형큐의 문제점을 해결한 큐. front와 rear의 값이 배열의 끝에 도달하면 다음에 증가되는 값은 0이 되도록 한다. ⇒ 처음과 끝이 연결되어 있다고 생각 !
→ 원형큐에서는 front와 rear의 초기값은 -1이 아닌 0으로 설정한다. front는 항상 큐의 첫 번째 요소의 하나 앞, rear는 마지막 요소를 가리킨다. 데이터를 삽입할 때는 rear를 먼저 증가시키고, 증가된 위치에 새로운 데이터를 삽입한다. 삭제할 때는 front를 먼저 증가시키고, 증가된 위치에서 데이터를 삭제한다.
→ 원형큐에서는 <span color="yellow_bg">하나의 자리는 항상 비워</span>둬야 하는데, 공백 상태와 포화 상태를 구별하기 위해서이다. 비워두지 않는다면 공백과 포화 둘 다 front == rear이기 때문에 구분을 할 수 없다. 하지만 요소들의 개수를 저장하고 있는 추가적인 변수가 있다면 한자리를 비워두지 않다도 된다.
→ 공백 상태 : front == rear / 포화 상태 : front = rear - 1

### 원형큐 구현
```c
#include <stdio.h>
#include <stdlib.h>

#define MAX_SIZE 5

typedef int element;
typedef struct {
    element data[MAX_SIZE];
    int front;
    int rear;
}Queue;
// 오류 함수
void error(char *message) {
    fprintf(stderr, "%s\n", message);
    exit(1);
}
// 큐 초기화
void init_queue(Queue *q) {
    q->front = 0;
    q->rear = 0;
}
// 공백 상태 검출 함수
int is_empty(Queue *q) {
    return (q->front == q->rear);
}
// 포화 상태 검출 함수
int is_full(Queue *q) {
    return ((q->rear + 1) % MAX_SIZE == q->front);
}
// 원형큐 출력 함수
void queue_print(Queue *q) {
printf("QUEUE(front=%d rear=%d) = ", q->front, q->rear);
if (!is_empty(q)) {
int i = q->front;
do {
i = (i + 1) % (MAX_SIZE);
printf("%d | ", q->data[i]);
if (i == q->rear) break;
} while (i != q->front);
}
printf("\n");
}
// 삽입 함수
void enqueue(Queue *q, element item) {
    if (is_full(q)) {
        error("queue is full");
    }
    else {
        q->rear = (q->rear + 1) % MAX_SIZE;
        q->data[q->rear] = item;
    }
}
// 삭제 함수
element dequeue(Queue *q) {
    if (is_empty(q)) {
        error("queue is empty");
    }
    else {
        q->front = (q->front + 1) % MAX_SIZE;
        return q->data[q->front];
    }
}
// 피크 함수
element peek(Queue *q) {
    if (is_empty(q)) {
        error("queue is empty");
    }
    else {
        return q->data[(q->front + 1) % MAX_SIZE];
    }
}
```

# 덱 ?
> deque(double-ended queue) : 큐의 front와 rear에서 모두 삽입과 삭제가 가능한 큐.

## 덱 ADT
```c
- 객체 : n개의 element형의 요소들의 순서 있는 모임
- 연산 :
create() ::= 덱을 생성한다.
init(dq) ::= 덱을 초기화한다.
is_empty(dq) ::= 덱이 공백 상태인지 검사한다.
is_full(dq) ::= 덱이 포화 상태인지 검사한다.
add_front(dq, e) ::= 덱의 앞에 요소를 초가한다.
add_rear(dq, e) ::= 덱의 뒤에 요소를 추가한다.
delete_front(dq) ::= 덱의 앞에 있는 요소를 반환한 다음 삭제한다.
delete_rear(dq) ::= 덱의 뒤에 있는 요소를 반환한 다음 삭제한다.
get_front(q) ::= 덱의 앞에서 삭제하지 않고 앞에 있는 요소를 반환한다.
get_rear(q) ::= 덱의 뒤에서 삭제하지 않고 뒤에 있는 요소를 반환한다.
```
→ 덱은 스택과 큐의 연산을 다 갖고 있다. add_front ↔ push , delete_front ↔ pop / add_rear ↔ enqueue , delete_front ↔ dequeue
→ 덱과 원형큐는 공통점이 많아 원형큐의 연산에서 delete_rear(), add_front(), get_rear() 연산을 추가하면 덱의 구현도 쉽게 가능하다.

### 덱 구현
![](/images/a32890ab6568b883.png)

```c
#include <stdio.h>
#include <stdlib.h>

#define MAX_SIZE 5

typedef int element;
typedef struct {
    element data[MAX_SIZE];
    int front;
    int rear;
}Deque;
// 오류 함수
void error(char *message) {
    fprintf(stderr, "%s\n", message);
    exit(1);
}
// 덱 초기화
void init_queue(Deque *q) {
    q->front = 0;
    q->rear = 0;
}
// 공백 상태 검출 함수
int is_empty(Deque *q) {
    return (q->front == q->rear);
}
// 포화 상태 검출 함수
int is_full(Deque *q) {
    return ((q->rear + 1) % MAX_SIZE == q->front);
}
// 덱 출력 함수
void queue_print(Deque *q) {
printf("QUEUE(front=%d rear=%d) = ", q->front, q->rear);
if (!is_empty(q)) {
int i = q->front;
do {
i = (i + 1) % (MAX_SIZE);
printf("%d | ", q->data[i]);
if (i == q->rear) break;
} while (i != q->front);
}
printf("\n");
}
// 삽입 함수
void add_rear(Deque *q, element item) {
    if (is_full(q)) error("Queue is full");
    q->rear = (q->rear + 1) % MAX_SIZE;
    q->data[q->rear] = item;
}
// 삭제 함수
element delete_front(Deque *q) {
    if (is_empty(q)) error("Queue is empty");
    q->front = (q->front + 1) % MAX_SIZE;
    return q->data[q->front];
}
// 덱의 앞쪽에 삽입하는 함수
void add_front(Deque *q, element item) {
    if (is_full(q)) error("Queue is full");
    q->data[q->front] = item;
    q->front = (q->front - 1 + MAX_SIZE) % MAX_SIZE;
}
// 덱의 뒤쪽에서 삭제하는 함수
element delete_rear(Deque *q) {
    if (is_empty(q)) error("Queue is empty");
    int prev = q->rear;
    q->rear = (q->rear - 1 + MAX_SIZE) % MAX_SIZE;
    return q->data[prev];
}
element get_rear(Deque *q) {
    if (is_empty(q)) error("Queue is empty");
    return q->data[q->rear];
}
```
