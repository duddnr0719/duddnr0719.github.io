## 1. 다음의 파일에 대하여 다음 물음에 답하시오.
- 10, 40, 30, 5, 12, 6, 15, 9, 60
### (1) 위의 파일을 순차적으로 읽어서 최소 히프트리를 구성하라. 공백 트리에서 최소 히프트리가 만들어지는 과정을 출력하라.
```c
#include <stdio.h>
#include <stdlib.h>

#define MAX_ELEMENT 200

// 힙 구조체 정의
typedef struct {
    int key;
} element;

typedef struct {
    element heap[MAX_ELEMENT];
    int heap_size; 
} HeapType;

// 힙 생성 함수
HeapType* create() {
    return (HeapType*)malloc(sizeof(HeapType));
}

// 힙 초기화 함수
void init(HeapType* h) {
    h->heap_size = 0;
}

// 최소 힙 삽입 함수
void insert_min_heap(HeapType *h, element item) {
    int i;
    i = ++(h->heap_size); // 새 원소를 삽입 하기 전에 힙 크기 증가하여 메모리 공간 확보

    while ((i != 1) && (item.key < h->heap[i/2].key)) { // 루트 노드가 아니고, 부모 노드의 키 값이 새 원소의 키 값보다 작을 때
        h->heap[i] = h->heap[i/2]; // 부모 노드를 자식 노드로 이동
        i /= 2; // i를 부모 노드의 인덱스로 변경
    }
    h->heap[i] = item; // 새 원소를 삽입
}

// 힙 출력 함수
void print_heap(HeapType* h) {
    for (int i = 1; i <= h->heap_size; i++) {
        printf(" %d->", h->heap[i].key);
    }
    printf("\n");
}

int main(void) {
    HeapType* h = create();
    init(h);

    // 힙 원소 정의
    element e1 = {10}, e2 = {40}, e3 = {30}, e4 = {5}, e5 = {12}, e6 = {6}, e7 = {15}, e8 = {9}, e9 = {60};
    insert_min_heap(h, e1);
    print_heap(h);
    insert_min_heap(h, e2);
    print_heap(h);
    insert_min_heap(h, e3);
    print_heap(h);
    insert_min_heap(h, e4);
    print_heap(h);
    insert_min_heap(h, e5);
    print_heap(h);
    insert_min_heap(h, e6);
    print_heap(h);
    insert_min_heap(h, e7);
    print_heap(h);
    insert_min_heap(h, e8);
    print_heap(h);
    insert_min_heap(h, e9);
    print_heap(h);
}
```

### (2)
1. 개요
  - Max Heap을 배열을 이용하여 구현하고 삽입과 삭제 연산을 구현한다.
1. 데이터는 정수형 값이 입력된다.
  1. Heap의 최대 크기는 100으로 정한다.
  1. 아래에 제시한 형태로 입력을 받으며, 요구한 형태로 출력을 해야 한다.
1. 기능
  1. 연산의 종류
    1. 삽입 (I) : max heap에 x를 삽입
    1. 삭제 (D) : 삭제 후 값 출력
  1. 예외 처리
    1. Empty : 비어있는 경우
    1. Full : 모두 차 있는 경우
  1. 종료 :  Q 입력
```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

#define MAX_ELEMENT 100

// 힙 구조체 정의
typedef struct {
    int key;
} element;

typedef struct {
    element heap[MAX_ELEMENT];
    int heap_size; 
} HeapType;

// 힙 생성 함수
HeapType* create() {
    return (HeapType*)malloc(sizeof(HeapType));
}

// 힙 초기화 함수
void init(HeapType* h) {
    h->heap_size = 0;
}

// 최대 힙 삽입 함수
void insert_max_heap(HeapType *h, element item) {
    int i;
    i = ++(h->heap_size); // 새 원소를 삽입 하기 전에 힙 크기 증가하여 메모리 공간 확보

    while ((i != 1) && (item.key > h->heap[i/2].key)) { // 루트 노드가 아니고, 부모 노드의 키 값이 새 원소의 키 값보다 클 때
        h->heap[i] = h->heap[i/2]; // 부모 노드를 자식 노드로 이동
        i /= 2; // i를 부모 노드의 인덱스로 변경
    }
    h->heap[i] = item; // 새 원소를 삽입
}

// 삭제 함수
element delete_max_heap(HeapType* h) {
    int parent, child; // 삭제할 노드의 부모 노드와 자식 노드의 인덱스 
    element item, temp; // 삭제할 노드와 마지막 노드

    item = h->heap[1]; // 루트 노드 저장
    temp = h->heap[(h->heap_size)--]; // 마지막 노드를 temp에 저장하고 힙 크기 감소
    parent = 1; // 루트 노드부터 시작
    child = 2; // 왼쪽 자식 노드부터 시작

    while (child <= h->heap_size) { // 자식 노드가 힙 크기보다 작거나 같을 때까지 반복
        if ((child < h->heap_size) && (h->heap[child].key < h->heap[child + 1].key)) child++; // 오른쪽 자식 노드가 더 크면 child를 오른쪽 자식 노드로 변경
        if (temp.key >= h->heap[child].key) break; // 부모 노드가 자식 노드보다 크면 종료
        h->heap[parent] = h->heap[child]; // 자식 노드를 부모 노드로 이동
        parent = child; // 부모 노드를 자식 노드로 변경
        child *= 2; // 자식 노드를 왼쪽 자식 노드로 변경
    }
    h->heap[parent] = temp; // 마지막 노드를 부모 노드로 이동
    return item;
}

// 힙 출력 함수
void print_heap(HeapType* h) {
    for (int i = 1; i <= h->heap_size; i++) {
        printf(" %d->", h->heap[i].key);
    }
    printf("\n");
}

int main(void) {
    HeapType* h = create();
    init(h);

    char select; // 사용자가 선택할 메뉴 저장할 변수
    bool flag = true; // 깃발을 세워 반복문 통제

    // 무한 반복을 통해 사용자가 종료를 원할 때까지 프로그램 실행
    while(flag) {
        printf("삽입(I), 삭제(D), 종료(Q): ");
        scanf(" %c", &select);
        element item;

        switch(select) { // 사용자로부터 원하는 메뉴 입력받아 분기문 실행  
            // 삽입의 경우
            case 'I': 
            printf("삽입할 key값: ");
            scanf("%d", &item.key);
            insert_max_heap(h, item);
            print_heap(h);
            break;

            // 삭제의 경우
            case 'D':
            // 힙 사이즈가 0인 경우 -> 힙 트리가 비어있는 것이기 때문에 오류 메세지 출력
            if (h->heap_size == 0) {
                printf("Empty.\n");
            } 
            // 힙 사이크가 MAX_ELEMENT를 초과하는 경우 -> 힙 트리가 가득 찬 것이기 때문에 오류 메세지 출력
            else if (h->heap_size > MAX_ELEMENT) {
                printf("Full.\n");
            }
            // 정상적인 경우
            else {
                item = delete_max_heap(h);
                printf("%d\n", item.key);
            }
            break;

            // 종료의 경우
            case 'Q':
            flag = false;
            break;
        }
    }
}
```
