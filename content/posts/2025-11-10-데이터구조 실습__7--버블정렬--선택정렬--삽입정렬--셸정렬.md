## 1. 개요: 입력 데이터를 정렬한다
### 2. 버블 정렬, 선택 정렬, 삽입정렬, 셀정렬을 구현한다.
### 3. 이번 과제는 2가지이다.
### 4. \[문제1\] 정렬단계 출력
1. 데이터는 랜덤함수를 이용하여 생성
1. 데이터의 범위는 \[1, 100\]
2. 데이터의 개수는 10개
3. 프로그램 구현 결과를 출력할 때 정렬이 되는 단계(step)를 보이도록 한다.
(예)
-----------------------
버블정렬
입력데이터: 9 5 8 4 1
Step 1: 5 9 8 4 1
Step 2: 5 8 9 4 1
…
정렬데이터: 1 4 5 8 9
------------------------
선택정렬
…
```c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <string.h>
#define SWAP(x, y, t) ( (t)=(x), (x)=(y), (y)=(t))

void print_array(int list[], int n) {
    for (int i = 0; i < n; i++) {
        printf("%d ", list[i]);
    }
    printf("\n");
}

// 선택 정렬
void selection_sort(int list[], int n)
{
    int i, j, least, temp;
    int step = 1;
    for (i=0; i < n-1; i++) {
        least = i;
        for (j=i+1; j<n; j++) // 최소값 탐색
            if(list[j] < list[least]) least = j;
        SWAP(list[i], list[least], temp);
    }
    printf("  Step %d: ", step++);
    print_array(list, n);
}

// 삽입 정렬
void insertion_sort(int list[], int n)
{
    int i, j, key;
    int step = 1;
    for (i=1; i<n; i++) {
        key = list[i];
        for (j=i-1; j>=0 && list[j]>key; j--)
            list[j+1]=list[j]; /* 레코드의 오른쪽 이동 */
    }
    list[j+1] = key;
    printf("  Step %d: ", step++);
    print_array(list, n);
}   

// 버블 정렬
void bubble_sort(int list[], int n)
{
    int i, j, temp;
    int step = 1;
    for(i=n-1; i>0; i--) {
        for (j=0; j<i; j++) {
            /* 앞 뒤의 레코드를 비교한 후 교체 */
            if (list[j] > list[j+1])
                SWAP(list[j], list[j+1], temp);
        }
        printf("  Step %d: ", step++);
        print_array(list, n);
    }
}

// gap만큼 떨어진 요소들을 삽입 정렬
// 정렬의 범위는 first에서 last
void inc_insertion_sort(int list[], int first, int last, int gap)
{
    int i, j, key;
    for(i=first+gap; i<=last; i=i+gap) {
        key = list[i];
        for (j=i-gap; j>=first && key<list[j]; j=j-gap)
            list[j+gap] = list[j];
        list[j+gap]=key;
    }
}

// 셸 정렬
void shell_sort(int list[], int n) 
{
    int i, gap;
    int step = 1;
    for (gap=n/2; gap>0; gap=gap/2) {
        if ((gap % 2)==0) gap++;
        for (i=0; i<gap; i++)   // 부분 리스트의 개수는 gap
            inc_insertion_sort(list, i, n-1, gap);
    }
    printf("  Step %d: ", step++);
    print_array(list, n);
}

int main(void) {
    int SIZE = 10;
    int original_data[SIZE];
    int list[SIZE]; // 정렬에 사용할 배열

    srand((unsigned int)time(NULL));

    // 난수 생성
    for (int i = 0; i < SIZE; i++) { 
        original_data[i] = rand() % 100 + 1;
    }

    printf("원본 데이터: ");
    print_array(original_data, SIZE);

    // memcpy를 사용하여 동일한 초기 난수 데이터로 실행 -> 모든 정렬에서 초기 리스트가 같은 배열로 실행되도록 설정
    printf("\n-----------------------\n");
    printf("버블 정렬\n");
    memcpy(list, original_data, SIZE * sizeof(int)); 
    printf("입력 데이터: ");
    print_array(list, SIZE);
    bubble_sort(list, SIZE);
    printf("정렬 데이터: ");
    print_array(list, SIZE);

    printf("\n-----------------------\n");
    printf("선택 정렬\n");
    memcpy(list, original_data, SIZE * sizeof(int)); 
    printf("입력 데이터: ");
    print_array(list, SIZE);
    selection_sort(list, SIZE);
    printf("정렬 데이터: ");
    print_array(list, SIZE);

    printf("\n-----------------------\n");
    printf("삽입 정렬\n");
    memcpy(list, original_data, SIZE * sizeof(int)); 
    printf("입력 데이터: ");
    print_array(list, SIZE);
    insertion_sort(list, SIZE);
    printf("정렬 데이터: ");
    print_array(list, SIZE);

    printf("\n-----------------------\n");
    printf("쉘 정렬\n");
    memcpy(list, original_data, SIZE * sizeof(int));
    printf("입력 데이터: ");
    print_array(list, SIZE);
    shell_sort(list, SIZE);
    printf("정렬 데이터: ");
    print_array(list, SIZE);
    printf("-----------------------\n");

    return 0;
}
```

### 5. \[문제2\] 처리시간 계산
1. 데이터는 랜덤함수를 이용하여 생성
1. 데이터의 범위는 \[1, 1,000,000\]
2. 데이터의 개수는 50,000개
3. 구현 후 각 정렬 알고리즘의 수행 시간 비교

```c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <string.h>
#define SWAP(x, y, t) ( (t)=(x), (x)=(y), (y)=(t))

// 선택 정렬
void selection_sort(int list[], int n)
{
    int i, j, least, temp;
    int step = 1;
    for (i=0; i < n-1; i++) {
        least = i;
        for (j=i+1; j<n; j++) // 최소값 탐색
            if(list[j] < list[least]) least = j;
        SWAP(list[i], list[least], temp);
    }
}

// 삽입 정렬
void insertion_sort(int list[], int n)
{
    int i, j, key;
    int step = 1;
    for (i=1; i<n; i++) {
        key = list[i];
        for (j=i-1; j>=0 && list[j]>key; j--)
            list[j+1]=list[j]; /* 레코드의 오른쪽 이동 */
    }
    list[j+1] = key;
}   

// 버블 정렬
void bubble_sort(int list[], int n)
{
    int i, j, temp;
    int step = 1;
    for(i=n-1; i>0; i--) {
        for (j=0; j<i; j++) {
            /* 앞 뒤의 레코드를 비교한 후 교체 */
            if (list[j] > list[j+1])
                SWAP(list[j], list[j+1], temp);
        }
    }
}

// gap만큼 떨어진 요소들을 삽입 정렬
// 정렬의 범위는 first에서 last
void inc_insertion_sort(int list[], int first, int last, int gap)
{
    int i, j, key;
    for(i=first+gap; i<=last; i=i+gap) {
        key = list[i];
        for (j=i-gap; j>=first && key<list[j]; j=j-gap)
            list[j+gap] = list[j];
        list[j+gap]=key;
    }
}

// 셸 정렬
void shell_sort(int list[], int n) 
{
    int i, gap;
    int step = 1;
    for (gap=n/2; gap>0; gap=gap/2) {
        if ((gap % 2)==0) gap++;
        for (i=0; i<gap; i++)   // 부분 리스트의 개수는 gap
            inc_insertion_sort(list, i, n-1, gap);
    }
}

// 배열에 새로운 난수를 생성하는 함수
void generate_random_data(int list[], int n, int max_val) {
    printf("새로운 난수 %d개 생성 중... (범위 [1, %d])\n", n, max_val);
    for (int i = 0; i < n; i++) {
        list[i] = (rand() % max_val) + 1;
    }
}

int main(void) {
    int SIZE = 50000;
    int MAX_VAL = 1000000;

    int* list = (int*)malloc(SIZE * sizeof(int));
    if (list == NULL) {
        printf("메모리 할당 실패!\n");
        return 1;
    }

    // 난수 시드 초기화
    srand((unsigned int)time(NULL));

    clock_t start, end;
    double elapsed_time;

    printf("\n-----------------------\n");
    printf("쉘 정렬 (Shell Sort) 측정 시작...\n");
    generate_random_data(list, SIZE, MAX_VAL); // 1. 난수 생성
    start = clock();                           // 2. 시간 측정 시작
    shell_sort(list, SIZE);                    // 3. 정렬 실행
    end = clock();                             // 4. 시간 측정 종료
    elapsed_time = (double)(end - start) / CLOCKS_PER_SEC;
    printf("-> 쉘 정렬 수행 시간: %.4f 초\n", elapsed_time);

    printf("\n-----------------------\n");
    printf("버블 정렬 (Bubble Sort) 측정 시작...\n");
    generate_random_data(list, SIZE, MAX_VAL); 
    start = clock();
    bubble_sort(list, SIZE);
    end = clock();
    elapsed_time = (double)(end - start) / CLOCKS_PER_SEC;
    printf("-> 버블 정렬 수행 시간: %.4f 초\n", elapsed_time);

    printf("\n-----------------------\n");
    printf("선택 정렬 (Selection Sort) 측정 시작...\n");
    generate_random_data(list, SIZE, MAX_VAL); 
    start = clock();
    selection_sort(list, SIZE);
    end = clock();
    elapsed_time = (double)(end - start) / CLOCKS_PER_SEC;
    printf("-> 선택 정렬 수행 시간: %.4f 초\n", elapsed_time);

    printf("\n-----------------------\n");
    printf("삽입 정렬 (Insertion Sort) 측정 시작...\n");
    generate_random_data(list, SIZE, MAX_VAL); 
    start = clock();
    insertion_sort(list, SIZE);
    end = clock();
    elapsed_time = (double)(end - start) / CLOCKS_PER_SEC;
    printf("-> 삽입 정렬 수행 시간: %.4f 초\n", elapsed_time);

    printf("\n-----------------------\n");
    printf("모든 측정 완료.\n");

    // 할당된 메모리 해제
    free(list);

    return 0;

}
```
